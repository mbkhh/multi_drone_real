# Lightweight real-flight loggers for both no-TF controller packages.
# Run this once on every drone before starting a logger. This directory is on
# persistent storage and survives a normal shutdown/reboot.
mkdir -p "$HOME/swarm_flight_logs"

# Do not tail these files over Wi-Fi during flight. Stop the logger cleanly,
# wait for disk writes, and use a new test number instead of overwriting a run.

# swarm_single_no_tf test 1 - Drone 1
ros2 run swarm_logger logger --ros-args \
  -p drone_count:=3 \
  -p local_drone_id:=1 \
  -p drone_ids:="[1]" \
  -p monitor_vehicle_topics:=false \
  -p monitor_tf:=false \
  -p monitor_wifi_events:=true \
  -p print_interval:=10.0 \
  -p wifi_interface:=wlan0 \
  -r __node:=swarm_logger_d1 \
  > "$HOME/swarm_flight_logs/no_tf_test1_d1.log" 2>&1
